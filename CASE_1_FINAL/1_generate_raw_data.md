# импортируем в сессию библиотеки pandas и numpy, присваивая короткое имя для быстрого вызова
import pandas as pd
import numpy as np

# заходим в библиотеку numpy через короткий алиас np
# далее, через точку-шлюз, обращаемся к внутреннему подмодулу random, относящегося к библиотеке np, который содержит функции генерации случайных чисел
# через вторую точку-шлюз, обращаемся к методу seed(), который является одной из функций подмодуля random
# метод seed() делает генерацию чисел предсказуемыой - является "закладкой в книге", задает фиксированную стартовую позицию, и выдает одну и ту же последовательность (отсылает на один и тот же текст книги)
np.random.seed(42)

# объявляем числовую переменную-константу n_rows, в которую записываем значение 1000
# это наш "архитектурный план" — жестко заданный масштаб, определяющий, сколько строк данных мы создадим во всей последующей цепочке кода
n_rows = 1000

# создаем три изолированных списка (list) в квадратных скобках
# clean_prices — "чистые" цены (положительные целые числа)
# refunds — возвраты/убытки (отрицательные числа, имитирующие финансовый отток)
# garbage — "мусорные" данные для стресс-теста (пустота None, строка-ошибка 'null_error' и строка с пробелами '    ')
clean_prices = [1500, 2500, 5000, 12300]
refunds = [-3000, -4500]
garbage = [None, 'null_error', '    ']

# формируем генеральный массив данных amount_pool ("общий бассейн", откуда мы будем черпать значения)
amount_pool = clean_prices * 4 + refunds * 1 + garbage * 1
# знак умножения (*) для списков в Python не умножает числа
# Знак * для списков в Python работает как копировальный аппарат. Он берет элементы внутри квадратных скобок и повторяет их друг за другом указанное количество раз
# знак плюса (+) склеивает эти размноженные списки в один длинный поезд данных
# clean_prices * 4 берет список из 4 элементов и превращает его в один длинный список из 16 элементов:[1500, 2500, 5000, 12300, 1500, 2500, 5000, 12300, 1500, 2500, 5000, 12300, 1500, 2500, 5000, 12300]
# refunds * 1 оставляет список как есть (всего 2 элемента):[-3000, -4500]
# garbage * 1 тоже оставляет список без изменений (3 элемента):[None, 'null_error', '    ']
# В результате переменная amount_pool становится одним большим списком, в котором лежат 21 элемент


# создаем структуру данных "словарь" (dictionary) через фигурные скобки и записываем её в переменную data
# словарь устроен по принципу "ключ: значение", где каждый ключ (например, 'tx_id') станет будущим названием колонки, а значение — её содержимым
data = {
    # КЛЮЧ 'tx_id' (ID транзакции): через встроенную функцию range() генерируем последовательность целых чисел
    # Функция range() в Python — это встроенный генератор (конвейер) арифметических прогрессий. Она выдает неизменяемую последовательность целых чисел.
    # Функция может принимать от одного до трех параметров: range(start, stop, step).
    # стартуем с 10001 и идем до (10001 + n_rows), то есть до 11001. Верхняя граница в range не включается, поэтому мы получим ровно 1000 уникальных ID по порядку
    # Pandas автоматически «прокрутит» этот конвейер от начала до конца, забирает все 1000 сгенерированных чисел и раскладывает их по строкам сверху вниз.
    'tx_id': range(10001, 10001 + n_rows),
    
    # КЛЮЧ 'user_id' (ID пользователя): ныряем через "точки-шлюзы" np -> random -> к функции randint()
    # randint(501, 550) генерирует случайные целые числа в диапазоне от 501 до 549 включительно (верхняя граница 550 исключается, отсюда ровно 50 уникальных ID клиентов)
    # через параметр size=n_rows приказываем функции выдать не одно число, а сразу заполнить массив длиной в 1000 элементов
    'user_id': np.random.randint(501, 550, size=n_rows), 
    
    # КЛЮЧ 'amount' (сумма): через "точки-шлюзы" np -> random вызываем функцию choice()
    # функция choice() работает как "лотерейный барабан": она принимает наш ранее созданный "бассейн" (amount_pool) и случайно вытаскивает из него элементы
    # параметр size=n_rows заставляет барабан прокрутиться ровно 1000 раз, чтобы на выходе получилась тысяча сумм с учетом заложенных нами перекосов и "мусора"
    'amount': np.random.choice(amount_pool, size=n_rows),
    
    # КЛЮЧ 'tx_date' (дата транзакции): снова используем "лотерейный барабан" choice() для выбора дат из предложенного списка строк
    # список намеренно "грязный": в нем смешаны разные форматы ISO ('2026-08-01'), европейский ('02.08.2026'), с косой чертой ('2026/08/03') и сломанный текст ('broken_date')
    # параметр size=n_rows раскидает эти текстовые даты по всем 1000 строкам будущей таблицы
    'tx_date': np.random.choice(
        ['2026-08-01', '02.08.2026', '2026/08/03', 'broken_date', '2026-08-05'],
        size=n_rows
    )
}

# обращаемся к библиотеке pandas через алиас pd, заходим через "точку-шлюз" в класс-конструктор DataFrame()
# передаем на вход наш структурированный словарь data. Pandas берет ключи и превращает их в шапку таблицы (столбцы), а массивы значений превращает в упорядоченные строки
# результат этой сборки упаковывается в переменную df_raw (сокращение от Dataframe Raw — сырая, необработанная таблица)
df_raw = pd.DataFrame(data)

# вызываем встроенную базовую функцию Python print() для вывода информации на экран
# передаем ей строковое значение (текст в кавычках), которое послужит визуальным разделителем и заголовком в консоли для последующих отчетов
print("--- ТЕХНИЧЕСКИЙ АУДИТ МАТРИЦЫ В RAM ---")
